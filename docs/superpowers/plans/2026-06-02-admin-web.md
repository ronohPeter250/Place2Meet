# Admin Web Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Firebase-backed `/admin` web system inside the existing Expo Router app for managing users and businesses.

**Architecture:** Add admin-only Expo routes under `app/admin`, with reusable admin services and UI under `src/admin`. The admin app uses Firebase Auth email/password, checks `admins/{uid}` authorization, writes Firestore-only moderation/status changes, and reuses the existing Google Places service for geocoded business addresses.

**Tech Stack:** Expo Router, React Native Web, TypeScript, Firebase Auth, Firestore, Storage, Google Places API, existing Place2Meet theme primitives.

```ts
import { Timestamp } from 'firebase/firestore';
import { AccountStatus } from './types';

export function normalizeSearchText(value: string | null | undefined): string {
  return (value ?? '').trim().toLowerCase();
}

export function calculateAgeFromBirthday(birthday: string | null | undefined): number | null {
  if (!birthday) return null;
  const birthDate = new Date(birthday);
  if (Number.isNaN(birthDate.getTime())) return null;
  const today = new Date();
  let age = today.getFullYear() - birthDate.getFullYear();
  const monthDiff = today.getMonth() - birthDate.getMonth();
  if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < birthDate.getDate())) age--;
  return age >= 0 && age <= 130 ? age : null;
}

export function getEffectiveAccountStatus(status?: AccountStatus): AccountStatus {
  return status ?? 'active';
}

export function isActiveAccount(status?: AccountStatus): boolean {
  return getEffectiveAccountStatus(status) === 'active';
}

export function accountStatusLabel(status?: AccountStatus): string {
  switch (getEffectiveAccountStatus(status)) {
    case 'active': return 'פעיל';
    case 'suspended': return 'מושהה';
    case 'blocked': return 'חסום';
    case 'deleted': return 'מחוק';
  }
}

export function formatTimestamp(value: unknown): string {
  if (!value) return '—';
  const date = value instanceof Timestamp
    ? value.toDate()
    : typeof value === 'object' && value !== null && 'toDate' in value && typeof value.toDate === 'function'
      ? value.toDate()
      : null;
  if (!date || Number.isNaN(date.getTime())) return '—';
  return new Intl.DateTimeFormat('he-IL', { dateStyle: 'medium', timeStyle: 'short' }).format(date);
}

export function includesSearch(source: string | null | undefined, query: string): boolean {
  const q = normalizeSearchText(query);
  if (!q) return true;
  return normalizeSearchText(source).includes(q);
}
```

```ts
import {
  addDoc,
  collection,
  doc,
  getDoc,
  getDocs,
  limit,
  orderBy,
  query,
  serverTimestamp,
  setDoc,
  updateDoc,
  where,
  GeoPoint,
} from 'firebase/firestore';
import { signInWithEmailAndPassword, signOut, User } from 'firebase/auth';
import { auth, db } from '../config/firebase';
import { BusinessProfile } from '../services/business';
import { UserProfile } from '../services/users';
import {
  AccountStatus,
  AdminAuditAction,
  AdminAuditLog,
  AdminProfile,
  PhotoModerationStatus,
} from './types';

const AUDIT_LIMIT = 50;

export async function loginAdmin(email: string, password: string): Promise<User> {
  const credential = await signInWithEmailAndPassword(auth, email.trim(), password);
  return credential.user;
}

export async function logoutAdmin(): Promise<void> {
  await signOut(auth);
}

export async function getAdminProfile(uid: string): Promise<AdminProfile | null> {
  const snap = await getDoc(doc(db, 'admins', uid));
  if (!snap.exists()) return null;
  return { uid, ...snap.data() } as AdminProfile;
}

export async function assertActiveAdmin(uid: string): Promise<AdminProfile> {
  const profile = await getAdminProfile(uid);
  if (!profile || profile.active !== true) {
    throw new Error('Unauthorized admin');
  }
  return profile;
}

export async function listUsersForAdmin(maxResults = 500): Promise<UserProfile[]> {
  const snap = await getDocs(query(collection(db, 'users'), limit(maxResults)));
  return snap.docs.map(d => ({ uid: d.id, ...d.data() })) as UserProfile[];
}

export async function getUserForAdmin(uid: string): Promise<UserProfile | null> {
  const snap = await getDoc(doc(db, 'users', uid));
  return snap.exists() ? ({ uid: snap.id, ...snap.data() } as UserProfile) : null;
}

export async function listBusinessesForAdmin(maxResults = 500): Promise<BusinessProfile[]> {
  const snap = await getDocs(query(collection(db, 'businesses'), limit(maxResults)));
  return snap.docs.map(d => ({ uid: d.id, ...d.data() })) as BusinessProfile[];
}

export async function getBusinessForAdmin(uid: string): Promise<BusinessProfile | null> {
  const snap = await getDoc(doc(db, 'businesses', uid));
  return snap.exists() ? ({ uid: snap.id, ...snap.data() } as BusinessProfile) : null;
}

async function writeAuditLog(params: Omit<AdminAuditLog, 'createdAt'>): Promise<void> {
  await addDoc(collection(db, 'adminAuditLogs'), {
    ...params,
    createdAt: serverTimestamp(),
  });
}

function currentAdminIdentity() {
  const admin = auth.currentUser;
  if (!admin) throw new Error('Admin is not signed in');
  return { adminId: admin.uid, adminEmail: admin.email ?? undefined };
}

export async function getAuditLogsForTarget(targetType: 'user' | 'business', targetId: string): Promise<AdminAuditLog[]> {
  const q = query(
    collection(db, 'adminAuditLogs'),
    where('targetType', '==', targetType),
    where('targetId', '==', targetId),
    orderBy('createdAt', 'desc'),
    limit(AUDIT_LIMIT),
  );
  const snap = await getDocs(q);
  return snap.docs.map(d => ({ id: d.id, ...d.data() })) as AdminAuditLog[];
}

export async function updateUserStatus(uid: string, before: AccountStatus | undefined, status: AccountStatus): Promise<void> {
  const admin = currentAdminIdentity();
  await updateDoc(doc(db, 'users', uid), {
    accountStatus: status,
    updatedByAdminAt: serverTimestamp(),
    updatedByAdminId: admin.adminId,
  });
  await writeAuditLog({ ...admin, targetType: 'user', targetId: uid, action: 'status_change', before: { accountStatus: before ?? 'active' }, after: { accountStatus: status } });
}

export async function updateUserFlag(uid: string, before: boolean | undefined, flagged: boolean): Promise<void> {
  const admin = currentAdminIdentity();
  await updateDoc(doc(db, 'users', uid), {
    flagged,
    updatedByAdminAt: serverTimestamp(),
    updatedByAdminId: admin.adminId,
  });
  await writeAuditLog({ ...admin, targetType: 'user', targetId: uid, action: 'flag_change', before: { flagged: before === true }, after: { flagged } });
}

export async function updateUserPhotoModeration(uid: string, photoKey: string, status: PhotoModerationStatus): Promise<void> {
  const admin = currentAdminIdentity();
  await updateDoc(doc(db, 'users', uid), {
    [`photoModeration.${photoKey}`]: status,
    updatedByAdminAt: serverTimestamp(),
    updatedByAdminId: admin.adminId,
  });
  await writeAuditLog({ ...admin, targetType: 'user', targetId: uid, action: 'photo_moderation', after: { [photoKey]: status } });
}

export async function updateUserAdminNotes(uid: string, before: string | undefined, adminNotes: string): Promise<void> {
  const admin = currentAdminIdentity();
  await updateDoc(doc(db, 'users', uid), {
    adminNotes,
    updatedByAdminAt: serverTimestamp(),
    updatedByAdminId: admin.adminId,
  });
  await writeAuditLog({ ...admin, targetType: 'user', targetId: uid, action: 'admin_notes_update', before: { adminNotes: before ?? '' }, after: { adminNotes } });
}

export interface AdminBusinessInput {
  uid?: string;
  contactName: string;
  businessName: string;
  email: string;
  location: string;
  coordinates: GeoPoint;
  about: string;
  category: string;
  hours: BusinessProfile['hours'];
  photos: string[];
  profileComplete: boolean;
  accountStatus: AccountStatus;
}

export async function saveBusinessForAdmin(input: AdminBusinessInput): Promise<string> {
  const admin = currentAdminIdentity();
  const id = input.uid || doc(collection(db, 'businesses')).id;
  const ref = doc(db, 'businesses', id);
  const before = await getDoc(ref);
  const data = {
    uid: id,
    contactName: input.contactName.trim(),
    businessName: input.businessName.trim(),
    email: input.email.trim(),
    location: input.location,
    coordinates: input.coordinates,
    about: input.about.trim(),
    category: input.category.trim(),
    hours: input.hours,
    photos: input.photos,
    profileComplete: input.profileComplete,
    accountStatus: input.accountStatus,
    updatedByAdminAt: serverTimestamp(),
    updatedByAdminId: admin.adminId,
    ...(before.exists() ? {} : { createdAt: serverTimestamp() }),
  };
  await setDoc(ref, data, { merge: true });
  await writeAuditLog({
    ...admin,
    targetType: 'business',
    targetId: id,
    action: before.exists() ? 'business_update' : 'business_create',
    before: before.exists() ? before.data() : undefined,
    after: data,
  });
  return id;
}
```

```tsx
import { createContext, useCallback, useContext, useEffect, useMemo, useState } from 'react';
import { User } from 'firebase/auth';
import { auth } from '../config/firebase';
import { AdminProfile } from './types';
import { assertActiveAdmin, loginAdmin, logoutAdmin } from './services';

interface AdminAuthState {
  user: User | null;
  admin: AdminProfile | null;
  isLoading: boolean;
  error: string | null;
}

interface AdminAuthContextType extends AdminAuthState {
  signIn: (email: string, password: string) => Promise<void>;
  signOut: () => Promise<void>;
  refresh: () => Promise<void>;
}

const AdminAuthContext = createContext<AdminAuthContextType | null>(null);

export function AdminAuthProvider({ children }: { children: React.ReactNode }) {
  const [state, setState] = useState<AdminAuthState>({ user: null, admin: null, isLoading: true, error: null });

  const loadAdmin = useCallback(async (user: User | null) => {
    if (!user) {
      setState({ user: null, admin: null, isLoading: false, error: null });
      return;
    }
    try {
      const admin = await assertActiveAdmin(user.uid);
      setState({ user, admin, isLoading: false, error: null });
    } catch {
      await logoutAdmin().catch(() => {});
      setState({ user: null, admin: null, isLoading: false, error: 'אין הרשאת אדמין למשתמש הזה' });
    }
  }, []);

  useEffect(() => {
    const unsubscribe = auth.onAuthStateChanged((user) => {
      setState(prev => ({ ...prev, user, isLoading: true }));
      loadAdmin(user);
    });
    return unsubscribe;
  }, [loadAdmin]);

  const signIn = useCallback(async (email: string, password: string) => {
    setState(prev => ({ ...prev, isLoading: true, error: null }));
    try {
      const user = await loginAdmin(email, password);
      await loadAdmin(user);
    } catch {
      setState({ user: null, admin: null, isLoading: false, error: 'ההתחברות נכשלה או שאין הרשאת אדמין' });
    }
  }, [loadAdmin]);

  const signOut = useCallback(async () => {
    await logoutAdmin();
    setState({ user: null, admin: null, isLoading: false, error: null });
  }, []);

  const refresh = useCallback(async () => {
    await loadAdmin(auth.currentUser);
  }, [loadAdmin]);

  const value = useMemo(() => ({ ...state, signIn, signOut, refresh }), [state, signIn, signOut, refresh]);
  return <AdminAuthContext.Provider value={value}>{children}</AdminAuthContext.Provider>;
}

export function useAdminAuth(): AdminAuthContextType {
  const context = useContext(AdminAuthContext);
  if (!context) throw new Error('useAdminAuth must be used inside AdminAuthProvider');
  return context;
}
```
